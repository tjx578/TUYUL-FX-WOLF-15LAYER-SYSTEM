"""Portable source extraction: pure OCI validation helpers; no campaign entrypoint."""
from __future__ import annotations

import hashlib
import re
import tarfile
from pathlib import PurePosixPath
from typing import Any
from urllib.parse import parse_qs, unquote

def sha256_bytes(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def _reject_absolute_or_drive_path(value: str, *, kind: str) -> None:
    candidate = PurePosixPath(value)
    if candidate.is_absolute() or re.match(r"^[a-zA-Z]:", value):
        raise ValueError(f"unsafe absolute {kind}: {value}")


def _resolve_link_parts(
    initial_parts: list[str], target: str, *, member: tarfile.TarInfo
) -> list[str]:
    resolved_parts = list(initial_parts)
    for part in PurePosixPath(target).parts:
        if part in {"", ".", "/"}:
            continue
        if part == "..":
            if not resolved_parts:
                raise ValueError(
                    f"layer link target escapes synthetic container root: "
                    f"{member.name} -> {member.linkname}"
                )
            resolved_parts.pop()
            continue
        resolved_parts.append(part)
    return resolved_parts


def _validate_link_target(member_path: str, member: tarfile.TarInfo) -> None:
    if not (member.issym() or member.islnk()):
        return
    target = member.linkname
    if not target or "\x00" in target:
        raise ValueError(f"empty or invalid link target: {member.name}")
    if "\\" in target:
        raise ValueError(
            f"non-POSIX layer link target rejected: {member.name} -> {member.linkname}"
        )
    if member.islnk():
        _reject_absolute_or_drive_path(target, kind="hardlink target")
        _resolve_link_parts([], target, member=member)
        return
    if target.startswith("//"):
        raise ValueError(
            f"ambiguous absolute symbolic-link target: {member.name} -> {member.linkname}"
        )
    if PurePosixPath(target).is_absolute():
        _resolve_link_parts([], target, member=member)
        return
    _reject_absolute_or_drive_path(target, kind="symbolic-link target")
    _resolve_link_parts(
        list(PurePosixPath(member_path).parent.parts), target, member=member
    )


def normalize_layer_member(member: tarfile.TarInfo) -> str | None:
    name = member.name
    if "\x00" in name:
        raise ValueError("layer member path contains NUL")
    if name in {".", "./"}:
        if not member.isdir() or member.linkname:
            raise ValueError(f"wrong-type root layer entry: {name}")
        return None
    normalized = name.replace("\\", "/")
    if not normalized:
        raise ValueError("empty layer member path")
    while normalized.startswith("./"):
        normalized = normalized[2:]
    _reject_absolute_or_drive_path(normalized, kind="layer member path")
    candidate = PurePosixPath(normalized)
    if ".." in candidate.parts:
        raise ValueError(f"unsafe layer member path: {name}")
    path = str(candidate)
    if path in {"", "."}:
        raise ValueError(f"empty layer member path: {name}")
    _validate_link_target(path, member)
    return path


def apply_overlay_replacement(
    stores: tuple[dict[str, bytes], ...], path: str, *, is_directory: bool
) -> None:
    prefix = path.rstrip("/") + "/"
    for store in stores:
        store.pop(path, None)
        if not is_directory:
            stale_descendants = [key for key in store if key.startswith(prefix)]
            for key in stale_descendants:
                del store[key]


def scan_content(
    data: bytes,
    path: str,
    layer_digest: str,
    patterns: dict[str, re.Pattern[bytes]],
) -> list[dict[str, str]]:
    findings: list[dict[str, str]] = []
    for rule, pattern in patterns.items():
        for match in pattern.finditer(data):
            findings.append(
                {
                    "rule": rule,
                    "path": path,
                    "layer_digest": layer_digest,
                    "match_sha256": sha256_bytes(match.group(0)),
                }
            )
    return findings


def scan_layer_member_metadata(
    member: tarfile.TarInfo,
    path: str,
    layer_digest: str,
    patterns: dict[str, re.Pattern[bytes]],
    suspicious_basenames: set[str],
) -> list[dict[str, str]]:
    findings: list[dict[str, str]] = []
    if not member.isdir() and PurePosixPath(path).name in suspicious_basenames:
        findings.append(
            {
                "rule": "suspicious_secret_filename",
                "path": path,
                "layer_digest": layer_digest,
                "match_sha256": sha256_bytes(path.encode("utf-8")),
            }
        )
    if member.issym() or member.islnk():
        findings.extend(
            scan_content(
                path.encode("utf-8"),
                f"{path}#member-path",
                layer_digest,
                patterns,
            )
        )
        findings.extend(
            scan_content(
                member.linkname.encode("utf-8"),
                f"{path}#link-target",
                layer_digest,
                patterns,
            )
        )
    return findings


def whiteout_structure_violation(
    member: tarfile.TarInfo, path: str, data: bytes | None
) -> dict[str, Any] | None:
    base = PurePosixPath(path).name
    if base != ".wh..wh..opq" and not base.startswith(".wh."):
        return None
    reasons: list[str] = []
    if not member.isfile():
        reasons.append("WHITEOUT_NOT_REGULAR_FILE")
    elif data is None:
        reasons.append("WHITEOUT_CONTENT_UNREADABLE")
    elif data:
        reasons.append("WHITEOUT_NOT_EMPTY")
    if not reasons:
        return None
    return {
        "path": path,
        "member_type_hex": member.type.hex(),
        "declared_size": member.size,
        "observed_bytes": None if data is None else len(data),
        "reasons": reasons,
    }


def base_material_uri_allowed(uri: str, expected_digest_hex: str) -> bool:
    normalized = unquote(uri).lower()
    base, separator, query = normalized.partition("?")
    if not separator:
        return False
    match = re.fullmatch(
        r"pkg:docker/(?:(?:docker|index)\.io/library/)?python(?:@([^?]+))?",
        base,
    )
    if match is None:
        return False
    try:
        params = parse_qs(query, strict_parsing=True, keep_blank_values=True)
    except ValueError:
        return False
    if any(len(values) != 1 for values in params.values()):
        return False
    platform = params.get("platform", [None])[0]
    version = match.group(1)
    if platform != "linux/amd64":
        return False
    expected_digest = f"sha256:{expected_digest_hex}"
    if version is None:
        return (
            set(params) == {"digest", "platform"}
            and params["digest"][0] == expected_digest
        )
    return set(params) == {"platform"} and version in {"3.11-slim", expected_digest}


def validate_runtime_config(config: dict[str, Any]) -> dict[str, bool]:
    runtime_config = config.get("config", {})
    runtime_env = set(runtime_config.get("Env", []) or [])
    return {
        "entrypoint_unset": runtime_config.get("Entrypoint") in (None, []),
        "cmd_exact": runtime_config.get("Cmd") == ["python", "api_server.py"],
        "working_directory_exact": runtime_config.get("WorkingDir") == "/app",
        "user_exact": runtime_config.get("User") == "appuser",
        "environment_contract": {
            "PYTHONDONTWRITEBYTECODE=1",
            "PYTHONUNBUFFERED=1",
            "ENV=production",
            "PORT=8000",
        }.issubset(runtime_env),
    }
