from __future__ import annotations

import contextlib
import json
from datetime import UTC, datetime
from typing import Any, cast

from redis.asyncio import Redis

from schemas.trade_models import RiskMode, Trade, TradeLeg, TradeStatus

from infrastructure.redis_client import get_client
from storage.redis_client import redis_client

# Terminal statuses that should NOT appear in active index
_TERMINAL_STATUSES = {TradeStatus.CANCELLED, TradeStatus.CLOSED, TradeStatus.SKIPPED}

# Redis SET key for active trade index
ACTIVE_INDEX_KEY = "TRADE:ACTIVE_INDEX"


def _redis_to_str(value: Any) -> str:
    if isinstance(value, bytes):
        return value.decode()
    return str(value)


class TradeLedger:
    """Read-oriented trade ledger for API and WS consumers."""

    _memory_trades: dict[str, Trade] = {}

    def get_trade(self, trade_id: str) -> Trade | None:
        if trade_id in self._memory_trades:
            return self._memory_trades[trade_id]

        with contextlib.suppress(Exception):
            raw = redis_client.client.get(f"TRADE:{trade_id}")
            if isinstance(raw, str):
                trade = self._from_dict(json.loads(raw))
                self._memory_trades[trade_id] = trade
                return trade
        return None

    async def get_trade_async(self, trade_id: str) -> Trade | None:
        if trade_id in self._memory_trades:
            return self._memory_trades[trade_id]

        with contextlib.suppress(Exception):
            client = await get_client()
            raw = await client.get(f"TRADE:{trade_id}")
            if isinstance(raw, str):
                trade = self._from_dict(json.loads(raw))
                self._memory_trades[trade_id] = trade
                return trade
        return None

    def get_active_trades(self) -> list[Trade]:
        """Return active trades using SET index + MGET (O(K) where K=active count)."""
        trades: dict[str, Trade] = dict(self._memory_trades)

        with contextlib.suppress(Exception):
            # Try indexed path: SMEMBERS + MGET = 2 RTTs total
            trade_ids: set[str] = set()
            raw_ids_result: set[str] = redis_client.client.smembers(ACTIVE_INDEX_KEY)  # type: ignore
            raw_ids: set[str] = {_redis_to_str(tid) for tid in raw_ids_result}
            if raw_ids:
                trade_ids = raw_ids

            if trade_ids:
                ordered_ids = sorted(trade_ids)
                keys = [f"TRADE:{tid}" for tid in ordered_ids]
                values = redis_client.client.mget(keys)
                if isinstance(values, list):
                    stale_ids: list[str] = []
                    for tid, raw in zip(ordered_ids, values):  # type: ignore[assignment]
                        if not isinstance(raw, str):
                            stale_ids.append(tid)
                            continue
                        trade = self._from_dict(json.loads(raw))
                        if trade.status in _TERMINAL_STATUSES:
                            stale_ids.append(tid)
                            continue
                        trades[tid] = trade

                    # Lazily prune terminal entries from the index
                    if stale_ids:
                        with contextlib.suppress(Exception):
                            redis_client.client.srem(ACTIVE_INDEX_KEY, *stale_ids)
            else:
                # Fallback: SCAN for first-time migration or empty index
                for key in redis_client.client.scan_iter(match="TRADE:*", count=200):  # type: ignore[call-arg]
                    key_str = _redis_to_str(key)
                    trade_id = key_str.split(":", 1)[1]
                    raw = redis_client.client.get(key_str)
                    if isinstance(raw, str):
                        trade = self._from_dict(json.loads(raw))
                        trades[trade_id] = trade
                        if trade.status not in _TERMINAL_STATUSES:
                            with contextlib.suppress(Exception):
                                redis_client.client.sadd(ACTIVE_INDEX_KEY, trade_id)

        return [
            t for t in sorted(trades.values(), key=lambda x: x.updated_at, reverse=True)
            if t.status not in _TERMINAL_STATUSES
        ]

    async def get_active_trades_async(self) -> list[Trade]:
        """Async version using SET index + MGET."""
        trades: dict[str, Trade] = dict(self._memory_trades)

        with contextlib.suppress(Exception):
            client = await get_client()

            trade_ids = await self.get_active_trade_identifiers(client)

            if trade_ids:
                ordered_ids = sorted(trade_ids)
                keys = [f"TRADE:{tid}" for tid in ordered_ids]
                values = await client.mget(keys)
                stale_ids: list[str] = []
                for tid, raw in zip(ordered_ids, values):
                    if not isinstance(raw, str):
                        stale_ids.append(tid)
                        continue
                    trade = self._from_dict(json.loads(raw))
                    if trade.status in _TERMINAL_STATUSES:
                        stale_ids.append(tid)
                        continue
                    trades[tid] = trade

                if stale_ids:
                    with contextlib.suppress(Exception):
                        await client.srem(ACTIVE_INDEX_KEY, *stale_ids) # type: ignore
            else:
                async for key in client.scan_iter(match="TRADE:*"):  # type: ignore[attr-defined]
                    key_str: str = _redis_to_str(key)
                    trade_id = key_str.split(":", 1)[1]
                    raw = await client.get(key_str)
                    if isinstance(raw, str):
                        trade = self._from_dict(json.loads(raw))
                        trades[trade_id] = trade
                        if trade.status not in _TERMINAL_STATUSES:
                            with contextlib.suppress(Exception):
                                await client.sadd(ACTIVE_INDEX_KEY, trade_id) # type: ignore

        return [
            t for t in sorted(trades.values(), key=lambda x: x.updated_at, reverse=True)
            if t.status not in _TERMINAL_STATUSES
        ]

    async def get_active_trade_identifiers(self, client: Redis) -> set[str]:
        raw_ids_result = await client.smembers(ACTIVE_INDEX_KEY)  # type: ignore[assignment]
        ids_set: set[Any] = cast(set[Any], raw_ids_result) if isinstance(raw_ids_result, set) else set()
        return {_redis_to_str(tid) for tid in ids_set}

    def _from_dict(self, data: dict[str, Any]) -> Trade:
        legs_payload: list[dict[str, Any]] = data.get("legs") or []
        if not legs_payload:
            legs_payload = [{
                "leg": 1,
                "entry": float(data.get("entry_price", 0.0) or 0.0),
                "sl": float(data.get("stop_loss", 0.0) or 0.0),
                "tp": float(data.get("take_profit", 0.0) or 0.0),
                "lot": float(data.get("lot_size", 0.01) or 0.01),
                "status": data.get("status", "INTENDED"),
            }]

        legs = [TradeLeg(**leg) for leg in legs_payload]

        status_raw = str(data.get("status", "INTENDED"))
        try:
            status = TradeStatus(status_raw)
        except ValueError:
            status = TradeStatus.INTENDED

        risk_mode_raw = str(data.get("risk_mode", "FIXED"))
        try:
            risk_mode = RiskMode(risk_mode_raw)
        except ValueError:
            risk_mode = RiskMode.FIXED

        return Trade(
            trade_id=str(data.get("trade_id", "")),
            signal_id=str(data.get("signal_id", "")),
            account_id=str(data.get("account_id", "")),
            pair=str(data.get("pair", "")),
            direction=str(data.get("direction", "BUY")),
            status=status,
            risk_mode=risk_mode,
            total_risk_percent=float(data.get("total_risk_percent", 0.1) or 0.1),
            total_risk_amount=float(data.get("total_risk_amount", 0.01) or 0.01),
            legs=legs,
            created_at=_parse_dt(data.get("created_at")),
            updated_at=_parse_dt(data.get("updated_at")),
            close_reason=data.get("close_reason"),
            pnl=float(data["pnl"]) if data.get("pnl") is not None else None,
        )


def _parse_dt(value: str | None) -> datetime:
    if not value:
        return datetime.now(UTC)
    try:
        return datetime.fromisoformat(value.replace("Z", "+00:00"))
    except Exception:
        return datetime.now(UTC)
